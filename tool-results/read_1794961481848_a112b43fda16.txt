     3→/**
     4→ * AI-Powered Coding Interview Prep — single-page dashboard.
     5→ *
     6→ * 3-column layout (responsive — stacks on mobile):
     7→ *   - LEFT:    problem list (filter by difficulty + category),
     8→ *             "Start technical" + "Start behavioral" buttons,
     9→ *             "Recommended next" card.
    10→ *   - CENTER:  chat panel (AI interviewer + user input + hint button +
    11→ *             submit button), code editor (plain <textarea>), "Run tests"
    12→ *             button + per-test results panel.
    13→ *   - RIGHT:   per-session score card + breakdown + "Recommended next" link
    14→ *             + aggregate stats (sessions completed, avg score, weak spots).
    15→ *
    16→ * State is held in this single client component — no global store. The API
    17→ * routes (see src/app/api) handle all persistence + LLM calls.
    18→ */
    19→

    44→import { Card, CardContent, CardHeader, CardTitle, CardDescription, CardFooter } from "@/components/ui/card";
    45→import { Badge } from "@/components/ui/badge";
    46→import { Input } from "@/components/ui/input";
    47→import { Textarea } from "@/components/ui/textarea";

    49→import {
    50→  Select,
    51→  SelectContent,
    52→  SelectItem,
    53→  SelectTrigger,
    54→  SelectValue,

    63→// --- types ------------------------------------------------------------------
    64→
    65→type Difficulty = "easy" | "medium" | "hard";
    66→type Category =
    67→  | "arrays"
    68→  | "strings"
    69→  | "dp"
    70→  | "graph"
    71→  | "tree"
    72→  | "two-pointers"
    73→  | "binary-search";
    74→
    75→interface Problem {
    76→  id: string;
    77→  title: string;
    78→  slug: string;
    79→  difficulty: Difficulty;
    80→  category: Category;
    81→  statement: string;
    82→  signature: string;
    83→  hints: string[];
    84→}
    85→
    86→interface Message {
    87→  id?: string;
    88→  role: "user" | "assistant" | "system";
    89→  content: string;
    90→  createdAt?: string;
    91→}
    92→
    93→interface TestResult {
    94→  name: string;
    95→  passed: boolean;
    96→  expected: string;
    97→  actual: string;
    98→  error?: string;
    99→}
   100→
   101→interface TestRunResponse {
   102→  testRunId: string;
   103→  passedCount: number;
   104→  totalCount: number;
   105→  results: TestResult[];
   106→  compileError?: string;
   107→  timedOut?: boolean;
   108→}
   109→
   110→interface Session {
   111→  id: string;
   112→  type: "technical" | "behavioral";
   113→  status: "active" | "completed";
   114→  score?: number | null;
   115→  scoreBreakdown?: any;
   116→  feedback?: string | null;
   117→  complexityReported?: string | null;
   118→  hintsUsedCount: number;
   119→  behavioralQuestion?: string | null;
   120→  problem?: Problem | null;
   121→  messages: Message[];
   122→  testRuns: any[];
   123→}
   124→
   125→interface Stats {
   126→  sessionsCompleted: number;
   127→  avgScore: number;
   128→  hintsPerSessionAvg: number;
   129→  totalHintsUsed: number;
   130→  categoryFailCounts: Record<string, number>;
   131→  weakness: Array<{ category: string; count: number }>;
   132→  lastRecommendedSlug: string | null;
   133→}

   138→const CATEGORIES: Category[] = [

   140→  "strings",
   141→  "dp",
   142→  "graph",
   143→  "tree",
   144→  "two-pointers",

   147→
   148→const DIFFICULTY_COLOR: Record<Difficulty, string> = {
   149→  easy: "bg-emerald-100 text-emerald-800 border-emerald-300",
   150→  medium: "bg-amber-100 text-amber-800 border-amber-300",
   151→  hard: "bg-rose-100 text-rose-800 border-rose-300",
   152→};

   154→// --- page ------------------------------------------------------------------
   155→
   156→export default function Page() {
   157→  const [problems, setProblems] = React.useState<Problem[]>([]);
   158→  const [stats, setStats] = React.useState<Stats | null>(null);
   159→  const [loadingProblems, setLoadingProblems] = React.useState(true);
   160→  const [filterDiff, setFilterDiff] = React.useState<Difficulty | "all">("all");
   161→  const [filterCat, setFilterCat] = React.useState<Category | "all">("all");
   162→
   163→  const [activeSession, setActiveSession] = React.useState<Session | null>(null);
   164→  const [code, setCode] = React.useState<string>("");
   165→  const [chatInput, setChatInput] = React.useState<string>("");
   166→  const [sending, setSending] = React.useState(false);
   167→  const [runningTests, setRunningTests] = React.useState(false);
   168→  const [submitting, setSubmitting] = React.useState(false);
   169→  const [testResults, setTestResults] = React.useState<TestRunResponse | null>(null);
   170→  const [startingSession, setStartingSession] = React.useState(false);
   171→  const [recommended, setRecommended] = React.useState<any | null>(null);
   172→
   173→  const loadProblems = React.useCallback(async () => {
   174→    setLoadingProblems(true);
   175→    try {
   176→      const params = new URLSearchParams();
   177→      if (filterDiff !== "all") params.set("difficulty", filterDiff);
   178→      if (filterCat !== "all") params.set("category", filterCat);
   179→      const res = await fetch(`/api/problems?${params.toString()}`);
   180→      const json = await res.json();
   181→      setProblems(json.problems || []);
   182→    } catch (e) {
   183→      toast.error("Failed to load problems", { description: (e as Error).message });
   184→    } finally {
   185→      setLoadingProblems(false);
   186→    }
   187→  }, [filterDiff, filterCat]);
   188→
   189→  const loadStats = React.useCallback(async () => {
   190→    try {
   191→      const res = await fetch("/api/stats");
   192→      const json = await res.json();
   193→      setStats(json.stats || null);
   194→    } catch {
   195→      // ignore — stats card just stays empty
   196→    }
   197→  }, []);
   198→
   199→  React.useEffect(() => {
   200→    loadProblems();
   201→    loadStats();
   202→  }, [loadProblems, loadStats]);
   203→
   204→  const startTechnical = async (problem: Problem) => {
   205→    setStartingSession(true);
   206→    try {
   207→      const res = await fetch("/api/interviews", {
   208→        method: "POST",
   209→        headers: { "Content-Type": "application/json" },
   210→        body: JSON.stringify({ type: "technical", problemId: problem.id }),
   211→      });
   212→      const json = await res.json();
   213→      if (!res.ok) throw new Error(json.error || "Failed to start session");
   214→      setActiveSession({
   215→        id: json.session.id,
   216→        type: "technical",
   217→        status: "active",
   218→        problem: json.session.problem,
   219→        hintsUsedCount: 0,
   220→        messages: [
   221→          { role: "assistant", content: json.greeting },
   222→        ],
   223→        testRuns: [],
   224→      });
   225→      setCode(starterCode(json.session.problem));
   226→      setTestResults(null);
   227→      setRecommended(null);
   228→      toast.success(`Started technical interview: ${problem.title}`);
   229→    } catch (e) {
   230→      toast.error("Failed to start interview", { description: (e as Error).message });
   231→    } finally {
   232→      setStartingSession(false);
   233→    }
   234→  };
   235→
   236→  const startBehavioral = async () => {
   237→    setStartingSession(true);
   238→    try {
   239→      const res = await fetch("/api/interviews", {
   240→        method: "POST",
   241→        headers: { "Content-Type": "application/json" },
   242→        body: JSON.stringify({ type: "behavioral" }),
   243→      });
   244→      const json = await res.json();
   245→      if (!res.ok) throw new Error(json.error || "Failed to start session");
   246→      setActiveSession({
   247→        id: json.session.id,
   248→        type: "behavioral",
   249→        status: "active",
   250→        behavioralQuestion: json.session.behavioralQuestion,
   251→        hintsUsedCount: 0,
   252→        messages: [{ role: "assistant", content: json.greeting }],
   253→        testRuns: [],
   254→      });
   255→      setCode("");
   256→      setTestResults(null);
   257→      setRecommended(null);
   258→      toast.success("Started behavioral interview");
   259→    } catch (e) {
   260→      toast.error("Failed to start interview", { description: (e as Error).message });
   261→    } finally {
   262→      setStartingSession(false);
   263→    }
   264→  };
   265→
   266→  const sendMessage = async (requestHint = false) => {
   267→    if (!activeSession) return;
   268→    const content = requestHint ? "I'm stuck — please give me a hint." : chatInput.trim();
   269→    if (!content) return;
   270→    setSending(true);
   271→    setChatInput("");
   272→    // Optimistic user message
   273→    const optimistic: Message = { role: "user", content };
   274→    setActiveSession((s) =>
   275→      s ? { ...s, messages: [...s.messages, optimistic] } : s,
   276→    );
   277→    try {
   278→      const res = await fetch(`/api/interviews/${activeSession.id}/messages`, {
   279→        method: "POST",
   280→        headers: { "Content-Type": "application/json" },
   281→        body: JSON.stringify({ content, requestHint }),
   282→      });
   283→      const json = await res.json();
   284→      if (!res.ok) throw new Error(json.error || "Failed to send message");
   285→      setActiveSession((s) =>
   286→        s
   287→          ? {
   288→              ...s,
   289→              hintsUsedCount: s.hintsUsedCount + (json.hintServed ? 1 : 0),
   290→              messages: [...s.messages, { role: "assistant", content: json.reply }],
   291→            }
   292→          : s,
   293→      );
   294→      if (json.hintServed) {
   295→        toast.info(`Hint ${json.hintIndex + 1}/${json.hintsTotal}${json.hintsExhausted ? " (final)" : ""}`);
   296→      }
   297→    } catch (e) {
   298→      toast.error("Failed to get AI reply", { description: (e as Error).message });
   299→    } finally {
   300→      setSending(false);
   301→    }
   302→  };
   303→
   304→  const runTests = async () => {
   305→    if (!activeSession || activeSession.type !== "technical") return;
   306→    if (!code.trim()) {
   307→      toast.warning("Write some code first");
   308→      return;
   309→    }
   310→    setRunningTests(true);
   311→    try {
   312→      const res = await fetch(`/api/interviews/${activeSession.id}/run-tests`, {
   313→        method: "POST",
   314→        headers: { "Content-Type": "application/json" },
   315→        body: JSON.stringify({ code }),
   316→      });
   317→      const json = await res.json();
   318→      if (!res.ok) throw new Error(json.error || "Failed to run tests");
   319→      setTestResults(json);
   320→      if (json.passedCount === json.totalCount) {
   321→        toast.success(`All ${json.totalCount} tests passed`);
   322→      } else {
   323→        toast.warning(`${json.passedCount}/${json.totalCount} tests passed`);
   324→      }
   325→    } catch (e) {
   326→      toast.error("Failed to run tests", { description: (e as Error).message });
   327→    } finally {
   328→      setRunningTests(false);
   329→    }
   330→  };
   331→
   332→  const submit = async () => {
   333→    if (!activeSession) return;
   334→    const payload =
   335→      activeSession.type === "technical"
   336→        ? code.trim()
   337→        : chatInput.trim() || (activeSession.messages.find((m) => m.role === "user")?.content ?? "");
   338→    if (!payload) {
   339→      toast.warning(
   340→        activeSession.type === "technical"
   341→          ? "Write some code before submitting"
   342→          : "Type your answer (200-500 words) before submitting",
   343→      );
   344→      return;
   345→    }
   346→    setSubmitting(true);
   347→    try {
   348→      const res = await fetch(`/api/interviews/${activeSession.id}/submit`, {
   349→        method: "POST",
   350→        headers: { "Content-Type": "application/json" },
   351→        body: JSON.stringify({ code: payload }),
   352→      });
   353→      const json = await res.json();
   354→      if (!res.ok) throw new Error(json.error || "Failed to submit");
   355→      setActiveSession((s) =>
   356→        s
   357→          ? {
   358→              ...s,
   359→              status: "completed",
   360→              score: json.score,
   361→              scoreBreakdown: json.breakdown,
   362→              feedback: json.feedback,
   363→              complexityReported: json.footer
   364→                ? `time: ${json.footer.timeComplexity}, space: ${json.footer.spaceComplexity}`
   365→                : s.complexityReported,
   366→              messages:
   367→                s.type === "behavioral"
   368→                  ? [...s.messages, { role: "assistant", content: json.feedback }]
   369→                  : s.messages,
   370→            }
   371→          : s,
   372→      );
   373→      if (json.testResults) setTestResults(json.testResults);
   374→      setRecommended(json.recommendedNext);
   375→      toast.success(`Submitted — score: ${json.score}/100`);
   376→      loadStats();
   377→    } catch (e) {
   378→      toast.error("Failed to submit", { description: (e as Error).message });
   379→    } finally {
   380→      setSubmitting(false);
   381→    }
   382→  };
   383→
   384→  const resetSession = () => {
   385→    setActiveSession(null);
   386→    setCode("");
   387→    setChatInput("");
   388→    setTestResults(null);
   389→    setRecommended(null);
   390→  };
   391→
   392→  // --- render --------------------------------------------------------------
   393→
   394→  return (
   395→    <div className="min-h-screen flex flex-col bg-background">
   396→      <Header />
   397→      <main className="flex-1 w-full max-w-[1600px] mx-auto px-3 sm:px-6 py-4">
   398→        <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   399→          {/* LEFT COLUMN — dashboard */}
   400→          <aside className="lg:col-span-3 flex flex-col gap-4">
   401→            <Card>
   402→              <CardHeader className="pb-3">
   403→                <CardTitle className="text-base flex items-center gap-2">
   404→                  <ListChecks className="size-4" /> Problem Bank
   405→                </CardTitle>
   406→                <CardDescription className="text-xs">
   407→                  {problems.length} problems — pick one to start
   408→                </CardDescription>
   409→              </CardHeader>
   410→              <CardContent className="space-y-3 pt-0">
   411→                <div className="grid grid-cols-2 gap-2">
   412→                  <div>
   413→                    <Label className="text-[10px] uppercase tracking-wide text-muted-foreground">
   414→                      Difficulty
   415→                    </Label>
   416→                    <Select value={filterDiff} onValueChange={(v) => setFilterDiff(v as any)}>
   417→                      <SelectTrigger className="h-8 text-xs mt-1">
   418→                        <SelectValue placeholder="all" />
   419→                      </SelectTrigger>
   420→                      <SelectContent>
   421→                        <SelectItem value="all">all</SelectItem>
   422→                        {DIFFICULTIES.map((d) => (
   423→                          <SelectItem key={d} value={d}>{d}</SelectItem>
   424→                        ))}
   425→                      </SelectContent>
   426→                    </Select>
   427→                  </div>
   428→                  <div>
   429→                    <Label className="text-[10px] uppercase tracking-wide text-muted-foreground">
   430→                      Category
   431→                    </Label>
   432→                    <Select value={filterCat} onValueChange={(v) => setFilterCat(v as any)}>
   433→                      <SelectTrigger className="h-8 text-xs mt-1">
   434→                        <SelectValue placeholder="all" />
   435→                      </SelectTrigger>
   436→                      <SelectContent>
   437→                        <SelectItem value="all">all</SelectItem>
   438→                        {CATEGORIES.map((c) => (
   439→                          <SelectItem key={c} value={c}>{c}</SelectItem>
   440→                        ))}
   441→                      </SelectContent>
   442→                    </Select>
   443→                  </div>
   444→                </div>
   445→                <ScrollArea className="max-h-96 pr-2">
   446→                  <div className="space-y-1.5">
   447→                    {loadingProblems
   448→                      ? Array.from({ length: 5 }).map((_, i) => (
   449→                          <Skeleton key={i} className="h-12 w-full" />
   450→                        ))
   451→                      : problems.map((p) => (
   452→                          <button
   453→                            key={p.id}
   454→                            onClick={() => startTechnical(p)}
   455→                            disabled={startingSession}
   456→                            className={cn(
   457→                              "w-full text-left p-2 rounded-md border bg-card hover:bg-accent transition-colors",
   458→                              "disabled:opacity-60 disabled:cursor-not-allowed",
   459→                              activeSession?.problem?.id === p.id &&
   460→                                "ring-2 ring-primary/40",
   461→                            )}
   462→                          >
   463→                            <div className="flex items-center justify-between gap-2">
   464→                              <span className="text-sm font-medium truncate">
   465→                                {p.title}
   466→                              </span>
   467→                              <Badge
   468→                                variant="outline"
   469→                                className={cn(
   470→                                  "text-[10px] px-1.5 py-0",
   471→                                  DIFFICULTY_COLOR[p.difficulty],
   472→                                )}
   473→                              >
   474→                                {p.difficulty}
   475→                              </Badge>
   476→                            </div>
   477→                            <div className="text-[11px] text-muted-foreground mt-0.5">
   478→                              {p.category} &middot; {p.testCasesCountLabel ?? ""}
   479→                              {p.hints?.length ?? 0} hints
   480→                            </div>
   481→                          </button>
   482→                        ))}
   483→                    {!loadingProblems && problems.length === 0 && (
   484→                      <p className="text-xs text-muted-foreground py-2 text-center">
   485→                        No problems match your filter.
   486→                      </p>
   487→                    )}
   488→                  </div>
   489→                </ScrollArea>
   490→              </CardContent>
   491→            </Card>
   492→
   493→            <Card>
   494→              <CardHeader className="pb-3">
   495→                <CardTitle className="text-base flex items-center gap-2">
   496→                  <Brain className="size-4" /> Behavioral Mode
   497→                </CardTitle>
   498→                <CardDescription className="text-xs">
   499→                  AI asks a STAR-format behavioral question
   500→                </CardDescription>
   501→              </CardHeader>
   502→              <CardContent className="pt-0">
   503→                <Button
   504→                  className="w-full"
   505→                  onClick={startBehavioral}
   506→                  disabled={startingSession}
   507→                >
   508→                  Start behavioral interview
   509→                </Button>
   510→              </CardContent>
   511→            </Card>
   512→
   513→            {recommended && (
   514→              <RecommendedNextCard
   515→                rec={recommended}
   516→                onStart={() => {
   517→                  const p = problems.find((p) => p.slug === recommended.slug);
   518→                  if (p) {
   519→                    startTechnical(p);
   520→                  } else {
   521→                    // Load full problem then start.
   522→                    fetch(`/api/problems/${recommended.slug}`)
   523→                      .then((r) => r.json())
   524→                      .then((json) => {
   525→                        if (json.problem) startTechnical(json.problem);
   526→                      })
   527→                      .catch(() => toast.error("Failed to load recommended problem"));
   528→                  }
   529→                }}
   530→              />
   531→            )}
   532→          </aside>
   533→
   534→          {/* CENTER COLUMN — interview */}
   535→          <section className="lg:col-span-6 flex flex-col gap-4">
   536→            {activeSession ? (
   537→              <InterviewColumn
   538→                session={activeSession}
   539→                code={code}
   540→                setCode={setCode}
   541→                chatInput={chatInput}
   542→                setChatInput={setChatInput}
   543→                sending={sending}
   544→                runningTests={runningTests}
   545→                submitting={submitting}
   546→                onSend={() => sendMessage(false)}
   547→                onHint={() => sendMessage(true)}
   548→                onRunTests={runTests}
   549→                onSubmit={submit}
   550→                onReset={resetSession}
   551→                testResults={testResults}
   552→              />
   553→            ) : (
   554→              <EmptyStateColumn
   555→                startingSession={startingSession}
   556→                onStartBehavioral={startBehavioral}
   557→              />
   558→            )}
   559→          </section>
   560→
   561→          {/* RIGHT COLUMN — score + stats */}
   562→          <aside className="lg:col-span-3 flex flex-col gap-4">
   563→            <ScoreCard session={activeSession} recommended={recommended} onReset={resetSession} />
   564→            <StatsCard stats={stats} />
   565→          </aside>
   566→        </div>
   567→      </main>
   568→      <footer className="mt-auto border-t bg-card py-3 text-center text-xs text-muted-foreground">
   569→        AI-Powered Coding Interview Prep &mdash; MVP &middot; runs your code in a
   570→        sandboxed bun subprocess (5s timeout) &middot; see README for
   571→        production sandboxing guidance.
   572→      </footer>
   573→    </div>
   574→  );
   575→}

   578→
   579→function Header() {
   580→  return (
   581→    <header className="border-b bg-card sticky top-0 z-10">
   582→      <div className="max-w-[1600px] mx-auto px-3 sm:px-6 py-3 flex items-center justify-between">
   583→        <div className="flex items-center gap-2">
   584→          <Sparkles className="size-5 text-primary" />
   585→          <div>
   586→            <h1 className="text-base sm:text-lg font-semibold leading-tight">
   587→              AI Coding Interview Prep
   588→            </h1>
   589→            <p className="text-[11px] text-muted-foreground hidden sm:block">
   590→              Mock technical + behavioral interviews, real-time feedback, performance scoring.
   591→            </p>
   592→          </div>
   593→        </div>
   594→        <div className="flex items-center gap-2 text-[11px] text-muted-foreground">
   595→          <Badge variant="outline" className="text-[10px]">
   596→            <Code2 className="size-3 mr-1" /> TypeScript
