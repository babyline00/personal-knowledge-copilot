    73→function renderCitationsInline(content: string): React.ReactNode[] {
    74→  // Replace [^c1] markers with clickable chips linked to the citation block.
    75→  const parts: React.ReactNode[] = []
    76→  const re = /\[\^(c\d+)\]/g
    77→  let last = 0
    78→  let m: RegExpExecArray | null
    79→  while ((m = re.exec(content)) !== null) {
    80→    if (m.index > last) {
    81→      parts.push(<React.Fragment key={`t-${last}`}>{content.slice(last, m.index)}</React.Fragment>)
    82→    }
    83→    const id = m[1]
    84→    parts.push(
    85→      <a
    86→        key={`c-${m.index}`}
    87→        href={`#cite-${id}`}
    88→        className="text-primary text-xs align-super mx-0.5 px-1 rounded bg-primary/10 hover:bg-primary/20 no-underline"
    89→      >
    90→        [{id}]
    91→      </a>,
    92→    )
    93→    last = m.index + m[0].length
    94→  }
    95→  if (last < content.length) {
    96→    parts.push(<React.Fragment key={`t-${last}`}>{content.slice(last)}</React.Fragment>)
    97→  }
    98→  return parts
    99→}

   106→// ---------- force-directed graph simulation ----------
   107→// A tiny home-grown force simulation. We run a fixed number of ticks per
   108→// animation frame and stop when the system cools below a threshold so the
   109→// CPU doesn't spin forever. Repulsion uses a simplified inverse-square law
   110→// over all node pairs (O(n^2), fine for a few hundred notes). Edges apply a
   111→// spring force toward their natural length. A centering force keeps the
   112→// graph in view.
   113→
   114→interface SimNode {
   115→  id: string
   116→  x: number
   117→  y: number
   118→  vx: number
   119→  vy: number
   120→  title: string
   121→  source: string
   122→  degree: number
   123→}
   124→
   125→function simulate(
   126→  nodes: SimNode[],
   127→  edges: { source: string; target: string }[],
   128→  opts: {
   129→    width: number
   130→    height: number
   131→    iterations: number
   132→    repulsion: number
   133→    spring: number
   134→    springLength: number
   135→    centering: number
   136→    damping: number
   137→  },
   138→): SimNode[] {
   139→  const idToNode = new Map(nodes.map((n) => [n.id, n]))
   140→  const edgePairs = edges
   141→    .map((e) => ({ a: idToNode.get(e.source), b: idToNode.get(e.target) }))
   142→    .filter((e): e is { a: SimNode; b: SimNode } => !!e.a && !!e.b)
   143→
   144→  for (let iter = 0; iter < opts.iterations; iter++) {
   145→    // Repulsion: every pair pushes apart.
   146→    for (let i = 0; i < nodes.length; i++) {
   147→      for (let j = i + 1; j < nodes.length; j++) {
   148→        const a = nodes[i]
   149→        const b = nodes[j]
   150→        let dx = a.x - b.x
   151→        let dy = a.y - b.y
   152→        let dist2 = dx * dx + dy * dy
   153→        if (dist2 < 1) {
   154→          // jitter identical positions
   155→          dx = (Math.random() - 0.5) * 2
   156→          dy = (Math.random() - 0.5) * 2
   157→          dist2 = dx * dx + dy * dy + 1
   158→        }
   159→        const dist = Math.sqrt(dist2)
   160→        const force = opts.repulsion / dist2
   161→        const fx = (dx / dist) * force
   162→        const fy = (dy / dist) * force
   163→        a.vx += fx
   164→        a.vy += fy
   165→        b.vx -= fx
   166→        b.vy -= fy
   167→      }
   168→    }
   169→
   170→    // Spring attraction along edges.
   171→    for (const e of edgePairs) {
   172→      const dx = e.b.x - e.a.x
   173→      const dy = e.b.y - e.a.y
   174→      const dist = Math.sqrt(dx * dx + dy * dy) || 1
   175→      const force = opts.spring * (dist - opts.springLength)
   176→      const fx = (dx / dist) * force
   177→      const fy = (dy / dist) * force
   178→      e.a.vx += fx
   179→      e.a.vy += fy
   180→      e.b.vx -= fx
   181→      e.b.vy -= fy
   182→    }
   183→
   184→    // Centering + integrate with damping.
   185→    const cx = opts.width / 2
   186→    const cy = opts.height / 2
   187→    for (const n of nodes) {
   188→      n.vx += (cx - n.x) * opts.centering
   189→      n.vy += (cy - n.y) * opts.centering
   190→      n.vx *= opts.damping
   191→      n.vy *= opts.damping
   192→      n.x += n.vx
   193→      n.y += n.vy
   194→      // Hard clamp inside the canvas so nodes don't drift off-screen.
   195→      n.x = Math.max(20, Math.min(opts.width - 20, n.x))
   196→      n.y = Math.max(20, Math.min(opts.height - 20, n.y))
   197→    }
   198→  }
   199→
   200→  return nodes
   201→}

   264→  // ---------- import handlers ----------
   265→  async function ingestFiles(files: FileList | File[]) {
   266→    setImporting(true)
   267→    try {
   268→      const fd = new FormData()
   269→      const arr = Array.from(files)
   270→      let validCount = 0
   271→      let skipped: string[] = []
   272→      for (const f of arr) {
   273→        if (/\.(md|markdown|txt|pdf)$/i.test(f.name)) {
   274→          fd.append('files', f)
   275→          validCount += 1
   276→        } else {
   277→          skipped.push(f.name)
   278→        }
   279→      }
   280→      if (skipped.length > 0) {
   281→        toast({ title: `Skipped ${skipped.length} unsupported file(s)`, description: skipped.join(', '), variant: 'destructive' })
   282→      }
   283→      if (!validCount) {
   284→        toast({ title: 'No .md / .txt / .pdf files dropped', variant: 'destructive' })
   285→        return
   286→      }
   287→      const res = await fetch('/api/notes/import', { method: 'POST', body: fd })
   288→      const json = await res.json()
   289→      if (!res.ok) throw new Error(json.error || `HTTP ${res.status}`)
   290→      const summary = json.results
   291→        .map((r: { title: string; chunks: number; source: string }) => `${r.title} (${r.chunks})`)
   292→        .join(', ')
   293→      toast({
   294→        title: `Imported ${json.results.length} note${json.results.length === 1 ? '' : 's'}`,
   295→        description: `Indexed ${json.totalChunks} chunks. ${summary}`,
   296→      })
   297→      await refreshNotes()
   298→    } catch (e) {
   299→      toast({ title: 'Import failed', description: (e as Error).message, variant: 'destructive' })
   300→    } finally {
   301→      setImporting(false)
   302→    }
   303→  }

   369→  // ---------- chat ----------
   370→  async function sendMessage() {
   371→    const text = input.trim()
   372→    if (!text || streaming) return
   373→    setInput('')
   374→    const userMsg: ChatMessage = { id: `u-${Date.now()}`, role: 'user', content: text, citations: [] }
   375→    const pendingMsg: ChatMessage = { id: `a-${Date.now()}`, role: 'assistant', content: '', citations: [], pending: true }
   376→    setMessages((m) => [...m, userMsg, pendingMsg])
   377→    setStreaming(true)
   378→
   379→    try {
   380→      const res = await fetch('/api/chat', {
   381→        method: 'POST',
   382→        headers: { 'Content-Type': 'application/json' },
   383→        body: JSON.stringify({
   384→          message: text,
   385→          sessionId: sessionId,
   386→          filterNoteIds: filterNoteIds.size ? Array.from(filterNoteIds) : undefined,
   387→        }),
   388→      })
   389→      if (!res.ok || !res.body) {
   390→        const errText = await res.text().catch(() => 'request failed')
   391→        throw new Error(errText)
   392→      }
   393→      const sid = res.headers.get('X-Session-Id')
   394→      if (sid) setSessionId(sid)
   395→
   396→      const reader = res.body.getReader()
   397→      const decoder = new TextDecoder()
   398→      let buffer = ''
   399→      let assistantText = ''
   400→      let footerParsed: { sessionId?: string; citations: Citation[] } | null = null
   401→
   402→      while (true) {
   403→        const { done, value } = await reader.read()
   404→        if (done) break
   405→        buffer += decoder.decode(value, { stream: true })
   406→
   407→        const sepIdx = buffer.indexOf('\n\n---CITATIONS---\n')
   408→        let body = buffer
   409→        if (sepIdx !== -1) {
   410→          body = buffer.slice(0, sepIdx)
   411→          const footerRaw = buffer.slice(sepIdx + '\n\n---CITATIONS---\n'.length)
   412→          try {
   413→            footerParsed = JSON.parse(footerRaw)
   414→            if (footerParsed?.sessionId) setSessionId(footerParsed.sessionId)
   415→          } catch {
   416→            footerParsed = { citations: [] }
   417→          }
   418→        }
   419→        if (body !== assistantText) {
   420→          assistantText = body
   421→          setMessages((m) =>
   422→            m.map((msg) => (msg.id === pendingMsg.id ? { ...msg, content: assistantText } : msg))
   423→          )
   424→        }
   425→      }
   426→      setMessages((m) =>
   427→        m.map((msg) =>
   428→          msg.id === pendingMsg.id
   429→            ? {
   430→                ...msg,
   431→                pending: false,
   432→                content: assistantText,
   433→                citations: footerParsed?.citations ?? [],
   434→              }
   435→            : msg
   436→        )
   437→      )
   438→    } catch (e) {
   439→      setMessages((m) =>
   440→        m.map((msg) =>
   441→          msg.id === pendingMsg.id
   442→            ? { ...msg, pending: false, content: `⚠️ ${(e as Error).message}` }
   443→            : msg
   444→        )
   445→      )
   446→      toast({ title: 'Chat error', description: (e as Error).message, variant: 'destructive' })
   447→    } finally {
   448→      setStreaming(false)
   449→    }
   450→  }

   836→// ---------- graph view ----------
   837→function GraphView({
   838→  data,
   839→  loading,
   840→  onRefresh,
   841→  onNodeClick,
   842→}: {
   843→  data: GraphData | null
   844→  loading: boolean
   845→  onRefresh: () => void
   846→  onNodeClick: (noteId: string) => void
   847→}) {
   848→  const containerRef = React.useRef<HTMLDivElement>(null)
   849→  const [dims, setDims] = React.useState({ w: 800, h: 600 })
   850→  const [hovered, setHovered] = React.useState<string | null>(null)
   851→  const [selected, setSelected] = React.useState<string | null>(null)
   852→  const [tick, setTick] = React.useState(0) // re-render trigger
   853→
   854→  // Track container size for responsive SVG.
   855→  React.useEffect(() => {
   856→    if (!containerRef.current) return
   857→    const el = containerRef.current
   858→    const update = () => setDims({ w: el.clientWidth, h: Math.max(400, el.clientHeight) })
   859→    update()
   860→    const ro = new ResizeObserver(update)
   861→    ro.observe(el)
   862→    return () => ro.disconnect()
   863→  }, [])
   864→
   865→  // Run the force simulation whenever data or container size changes.
   866→  const simRef = React.useRef<SimNode[]>([])
   867→  React.useEffect(() => {
   868→    if (!data || !dims.w) return
   869→    const existing = new Map(simRef.current.map((n) => [n.id, n]))
   870→    const cx = dims.w / 2
   871→    const cy = dims.h / 2
   872→    const nodes: SimNode[] = data.nodes.map((n) => {
   873→      const prev = existing.get(n.id)
   874→      if (prev) {
   875→        return { ...prev, ...n, vx: prev.vx * 0.5, vy: prev.vy * 0.5 }
   876→      }
   877→      // Seed on a circle around the centre so the layout doesn't start
   878→      // collapsed.
   879→      const angle = Math.random() * Math.PI * 2
   880→      const radius = Math.min(dims.w, dims.h) * 0.3
   881→      return {
   882→        id: n.id,
   883→        title: n.title,
   884→        source: n.source,
   885→        degree: n.degree,
   886→        x: cx + Math.cos(angle) * radius,
   887→        y: cy + Math.sin(angle) * radius,
   888→        vx: 0,
   889→        vy: 0,
   890→      }
   891→    })
   892→    simRef.current = nodes
   893→    // Initial settle: simulate more iterations up-front so the graph appears
   894→    // already laid out on first paint instead of starting as a tight cluster.
   895→    simulate(nodes, data.edges, {
   896→      width: dims.w,
   897→      height: dims.h,
   898→      iterations: 220,
   899→      repulsion: 6000,
   900→      spring: 0.04,
   901→      springLength: 110,
   902→      centering: 0.01,
   903→      damping: 0.7,
   904→    })
   905→    setTick((t) => t + 1)
   906→  }, [data, dims.w, dims.h])
   907→
   908→  // Continuous animation — runs the simulation a few ticks per frame so the
   909→  // graph gently settles instead of jumping to the final layout.
   910→  React.useEffect(() => {
   911→    if (!data) return
   912→    let raf = 0
   913→    let running = true
   914→    let cooldown = 60 // frames of gentle animation, then we stop to save CPU
   915→    const tickFn = () => {
   916→      if (!running) return
   917→      if (cooldown > 0) {
   918→        simulate(simRef.current, data.edges, {
   919→          width: dims.w,
   920→          height: dims.h,
   921→          iterations: 6,
   922→          repulsion: 6000,
   923→          spring: 0.04,
   924→          springLength: 110,
   925→          centering: 0.01,
   926→          damping: 0.85,
   927→        })
   928→        cooldown -= 1
   929→        setTick((t) => (t + 1) % 1_000_000)
   930→      }
   931→      raf = requestAnimationFrame(tickFn)
   932→    }
   933→    raf = requestAnimationFrame(tickFn)
   934→    return () => {
   935→      running = false
   936→      cancelAnimationFrame(raf)
   937→    }
   938→  }, [data, dims.w, dims.h])
   939→
   940→  const nodes = simRef.current
   941→  if (loading) {
   942→    return (
   943→      <main className="flex-1 grid place-items-center p-8">
   944→        <div className="flex items-center gap-3 text-muted-foreground">
   945→          <Loader2 className="size-5 animate-spin" />
   946→          <span className="text-sm">Loading graph…</span>
   947→        </div>
   948→      </main>
   949→    )
   950→  }
   951→  if (!data || data.nodes.length === 0) {
   952→    return (
   953→      <main className="flex-1 grid place-items-center p-8">
   954→        <div className="text-center max-w-md">
   955→          <div className="size-14 rounded-full bg-primary/10 grid place-items-center mx-auto mb-4">
   956→            <Network className="size-7 text-primary" />
   957→          </div>
   958→          <h2 className="text-lg font-semibold mb-2">No graph yet</h2>
   959→          <p className="text-sm text-muted-foreground">
   960→            Import markdown notes containing <code>[[Wiki Links]]</code> to build the graph.
   961→            Notes that link to each other appear as connected nodes; clicking a node filters the chat to that note.
   962→          </p>
   963→          <Button variant="outline" size="sm" className="mt-4" onClick={onRefresh}>
   964→            Refresh graph
   965→          </Button>
   966→        </div>
   967→      </main>
   968→    )
   969→  }
   970→
   971→  // Map node id -> position for edge rendering.
   972→  const pos = new Map(nodes.map((n) => [n.id, n]))
   973→  const maxDegree = Math.max(1, ...data.nodes.map((n) => n.degree))
   974→
   975→  return (
   976→    <main className="flex-1 flex flex-col p-4 sm:p-6">
   977→      <div className="flex items-center justify-between mb-3">
   978→        <div>
   979→          <h2 className="text-sm font-semibold">Note Link Graph</h2>
   980→          <p className="text-xs text-muted-foreground">
   981→            {data.nodes.length} notes · {data.edges.length} links · click a node to filter chat
   982→          </p>
   983→        </div>
   984→        <Button variant="outline" size="sm" onClick={onRefresh}>
   985→          Refresh
   986→        </Button>
   987→      </div>
   988→      <div ref={containerRef} className="relative flex-1 rounded-xl border bg-card overflow-hidden">
   989→        <svg
   990→          width={dims.w}
   991→          height={dims.h}
   992→          className="block"
   993→          aria-label="Wikilink graph"
   994→          role="img"
   995→          style={{ display: 'block' }}
   996→        >
   997→          {/* Edges */}
   998→          <g>
   999→            {data.edges.map((e, i) => {
  1000→              const a = pos.get(e.source)
  1001→              const b = pos.get(e.target)
  1002→              if (!a || !b) return null
  1003→              const dim = hovered && hovered !== e.source && hovered !== e.target ? 0.15 : 0.5
  1004→              return (
  1005→                <line
  1006→                  key={`e-${i}`}
  1007→                  x1={a.x}
  1008→                  y1={a.y}
  1009→                  x2={b.x}
  1010→                  y2={b.y}
  1011→                  stroke="currentColor"
  1012→                  className="text-muted-foreground"
  1013→                  strokeOpacity={dim}
  1014→                  strokeWidth={1}
  1015→                />
  1016→              )
  1017→            })}
  1018→          </g>
  1019→          {/* Nodes */}
  1020→          <g>
  1021→            {nodes.map((n) => {
  1022→              const radius = 8 + (n.degree / maxDegree) * 18
  1023→              const isHover = hovered === n.id
  1024→              const isSel = selected === n.id
  1025→              const dim = hovered && hovered !== n.id ? 0.4 : 1
  1026→              return (
  1027→                <g
  1028→                  key={n.id}
  1029→                  transform={`translate(${n.x},${n.y})`}
  1030→                  className="cursor-pointer"
  1031→                  onMouseEnter={() => setHovered(n.id)}
  1032→                  onMouseLeave={() => setHovered(null)}
  1033→                  onClick={() => {
  1034→                    setSelected(n.id)
  1035→                    onNodeClick(n.id)
  1036→                  }}
  1037→                  opacity={dim}
  1038→                >
  1039→                  <circle
  1040→                    r={radius}
  1041→                    className={
  1042→                      isSel
  1043→                        ? 'fill-primary stroke-primary-foreground stroke-2'
  1044→                        : isHover
  1045→                        ? 'fill-primary/80 stroke-primary-foreground/60 stroke-1.5'
  1046→                        : 'fill-primary/70 stroke-background stroke-1.5'
  1047→                    }
  1048→                  />
  1049→                  <text
  1050→                    y={radius + 12}
  1051→                    textAnchor="middle"
  1052→                    className="fill-foreground text-[10px] pointer-events-none select-none"
  1053→                    style={{ font: '10px var(--font-geist-sans, sans-serif)' }}
  1054→                  >
  1055→                    {n.title.length > 28 ? n.title.slice(0, 25) + '…' : n.title}
  1056→                  </text>
  1057→                </g>
  1058→              )
  1059→            })}
  1060→          </g>
  1061→        </svg>
  1062→        {/* Hint */}
  1063→        <div className="absolute bottom-2 left-2 right-2 flex items-center justify-between text-[10px] text-muted-foreground/70 pointer-events-none">
  1064→          <span>Tip: click a node to filter chat · size = degree</span>
  1065→          <a
  1066→            href="https://github.com/"
  1067→            target="_blank"
  1068→            rel="noreferrer"
  1069→            className="pointer-events-auto inline-flex items-center gap-1 hover:text-foreground"
  1070→            onClick={(e) => e.preventDefault()}
  1071→          >
  1072→            <Github className="size-3" />
  1073→            source
  1074→          </a>
  1075→        </div>
  1076→      </div>
  1077→      {/* Legend */}
  1078→      <div className="mt-3 flex flex-wrap items-center gap-4 text-[11px] text-muted-foreground">
  1079→        <span className="flex items-center gap-1.5">
  1080→          <span className="inline-block size-3 rounded-full bg-primary/70" /> Note
  1081→        </span>
  1082→        <span className="flex items-center gap-1.5">
  1083→          <span className="inline-block w-6 h-px bg-muted-foreground" /> [[wikilink]]
  1084→        </span>
  1085→        <span>Node radius scales with the number of links.</span>
  1086→      </div>
  1087→      {/* silence unused warning */}
  1088→      <span className="hidden">{tick}</span>
  1089→    </main>
  1090→  )
