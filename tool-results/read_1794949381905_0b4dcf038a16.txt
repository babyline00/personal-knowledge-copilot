    67→// ---------- helpers ----------
    68→function renderCitationsInline(content: string): React.ReactNode[] {
    69→  // Replace [^c1] markers with clickable chips linked to the citation block.
    70→  const parts: React.ReactNode[] = []
    71→  const re = /\[\^(c\d+)\]/g
    72→  let last = 0
    73→  let m: RegExpExecArray | null
    74→  while ((m = re.exec(content)) !== null) {
    75→    if (m.index > last) {
    76→      parts.push(<React.Fragment key={`t-${last}`}>{content.slice(last, m.index)}</React.Fragment>)
    77→    }
    78→    const id = m[1]
    79→    parts.push(
    80→      <a
    81→        key={`c-${m.index}`}
    82→        href={`#cite-${id}`}
    83→        className="text-primary text-xs align-super mx-0.5 px-1 rounded bg-primary/10 hover:bg-primary/20 no-underline"
    84→      >
    85→        [{id}]
    86→      </a>,
    87→    )
    88→    last = m.index + m[0].length
    89→  }
    90→  if (last < content.length) {
    91→    parts.push(<React.Fragment key={`t-${last}`}>{content.slice(last)}</React.Fragment>)
    92→  }
    93→  return parts
    94→}

    96→function fmtDate(s: string): string {
    97→  const d = new Date(s)
    98→  return d.toLocaleString(undefined, { month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit' })
    99→}
   100→
   101→function fmtDuration(sec: number | null): string {
   102→  if (!sec || sec <= 0) return ''
   103→  return formatTimestamp(sec)
   104→}

   230→  // ---------- chat ----------
   231→  async function sendMessage() {
   232→    const text = input.trim()
   233→    if (!text || streaming) return
   234→    setInput('')
   235→    const userMsg: ChatMessage = { id: `u-${Date.now()}`, role: 'user', content: text, citations: [] }
   236→    const pendingMsg: ChatMessage = { id: `a-${Date.now()}`, role: 'assistant', content: '', citations: [], pending: true }
   237→    setMessages((m) => [...m, userMsg, pendingMsg])
   238→    setStreaming(true)
   239→
   240→    try {
   241→      const res = await fetch('/api/chat', {
   242→        method: 'POST',
   243→        headers: { 'Content-Type': 'application/json' },
   244→        body: JSON.stringify({
   245→          message: text,
   246→          sessionId: sessionId,
   247→          filterVideoIds: filterVideoIds.size ? Array.from(filterVideoIds) : undefined,
   248→        }),
   249→      })
   250→      if (!res.ok || !res.body) {
   251→        const errText = await res.text().catch(() => 'request failed')
   252→        throw new Error(errText)
   253→      }
   254→      const sid = res.headers.get('X-Session-Id')
   255→      if (sid) setSessionId(sid)
   256→
   257→      const reader = res.body.getReader()
   258→      const decoder = new TextDecoder()
   259→      let buffer = ''
   260→      let assistantText = ''
   261→      let footerParsed: { sessionId?: string; citations: Citation[]; activeVideoId?: string | null; activeStartSec?: number | null } | null = null
   262→
   263→      while (true) {
   264→        const { done, value } = await reader.read()
   265→        if (done) break
   266→        buffer += decoder.decode(value, { stream: true })
   267→
   268→        const sepIdx = buffer.indexOf('\n\n---CITATIONS---\n')
   269→        let body = buffer
   270→        if (sepIdx !== -1) {
   271→          body = buffer.slice(0, sepIdx)
   272→          const footerRaw = buffer.slice(sepIdx + '\n\n---CITATIONS---\n'.length)
   273→          try {
   274→            footerParsed = JSON.parse(footerRaw)
   275→            if (footerParsed?.sessionId) setSessionId(footerParsed.sessionId)
   276→          } catch {
   277→            footerParsed = { citations: [] }
   278→          }
   279→        }
   280→        if (body !== assistantText) {
   281→          assistantText = body
   282→          setMessages((m) =>
   283→            m.map((msg) => (msg.id === pendingMsg.id ? { ...msg, content: assistantText } : msg))
   284→          )
   285→        }
   286→      }
   287→      setMessages((m) =>
   288→        m.map((msg) =>
   289→          msg.id === pendingMsg.id
   290→            ? {
   291→                ...msg,
   292→                pending: false,
   293→                content: assistantText,
   294→                citations: footerParsed?.citations ?? [],
   295→              }
   296→            : msg
   297→        )
   298→      )
   299→
   300→      // Auto-focus the player on the FIRST cited video so the user can
   301→      // immediately see where the answer came from.
   302→      if (footerParsed?.activeVideoId) {
   303→        const vid = videos.find((v) => v.id === footerParsed.activeVideoId) ?? null
   304→        const firstCit = footerParsed.citations[0]
   305→        const youtubeId = vid?.youtubeId ?? firstCit?.videoId ?? null
   306→        if (youtubeId) {
   307→          // If the active video is the same as the cited video, just seek; else swap.
   308→          if (activeVideo?.id !== footerParsed.activeVideoId) {
   309→            setActiveVideo({
   310→              id: footerParsed.activeVideoId,
   311→              youtubeId,
   312→              title: vid?.title ?? firstCit?.videoTitle ?? 'YouTube video',
   313→              startSec: footerParsed.activeStartSec ?? undefined,
   314→            })
   315→          } else if (typeof footerParsed.activeStartSec === 'number') {
   316→            seekPlayer(footerParsed.activeStartSec)
   317→          }
   318→        }
   319→      }
   320→    } catch (e) {
   321→      setMessages((m) =>
   322→        m.map((msg) =>
   323→          msg.id === pendingMsg.id
   324→            ? { ...msg, pending: false, content: `⚠️ ${(e as Error).message}` }
   325→            : msg
   326→        )
   327→      )
   328→      toast({ title: 'Chat error', description: (e as Error).message, variant: 'destructive' })
   329→    } finally {
   330→      setStreaming(false)
   331→    }
   332→  }

   341→  // ---------- player seek ----------
   342→  // The right-column iframe uses the YouTube IFrame Player API. We postMessage
   343→  // the `seekTo` command to it. (The `enablejsapi=1` URL param activates the
   344→  // postMessage API.)
   345→  function seekPlayer(seconds: number) {
   346→    const iframe = playerRef.current
   347→    if (!iframe) return
   348→    const sec = Math.max(0, Math.floor(seconds))
   349→    // YouTube IFrame API: {"event":"command","func":"seekTo","args":["<sec>", true]}
   350→    iframe.contentWindow?.postMessage(
   351→      JSON.stringify({ event: 'command', func: 'seekTo', args: [String(sec), true] }),
   352→      '*',
   353→    )
   354→    iframe.contentWindow?.postMessage(
   355→      JSON.stringify({ event: 'command', func: 'playVideo', args: [] }),
   356→      '*',

   557→                        }}
   558→                        aria-label={`Delete ${v.title}`}
   559→                      >
   560→                        <Trash2 className="size-3.5" />
   561→                      </Button>
   562→                    </div>
   563→                  </Card>
   564→                ))
   565→              )}
   566→            </div>
   567→          </ScrollArea>
   568→        </aside>
   569→
   570→        {/* Center column: chat */}
   571→        <section className="flex flex-col min-h-[60vh] lg:h-[calc(100vh-57px)] border-r">
   572→          {/* Chat header */}
   573→          <div className="px-4 py-3 border-b flex items-center gap-2">
   574→            <MessageSquare className="size-4 text-muted-foreground" />
   575→            <h2 className="text-sm font-semibold">Ask across transcripts</h2>
   576→            {activeFilterCount > 0 && (
   577→              <Badge variant="secondary" className="text-[10px]">
   578→                {activeFilterCount} video{activeFilterCount === 1 ? '' : 's'} filtered
   579→              </Badge>
   580→            )}
   581→            <div className="ml-auto">
   582→              <Button
   583→                variant="ghost"
   584→                size="sm"
   585→                className="h-7 text-xs"
   586→                onClick={() => {
   587→                  setMessages([])
   588→                  setSessionId(null)
   589→                }}
   590→                disabled={!messages.length || streaming}
   591→              >
   592→                <X className="size-3.5 mr-1" />
   593→                Clear
   594→              </Button>
   595→            </div>
   596→          </div>
   597→
   598→          {/* Messages */}
   599→          <ScrollArea className="flex-1">
   600→            <div className="p-4 space-y-4 max-w-3xl mx-auto">
   601→              {messages.length === 0 ? (
   602→                <div className="text-center text-sm text-muted-foreground py-16">

   613→                    <div
   614→                      className={`rounded-lg px-4 py-3 max-w-[85%] ${
   615→                        m.role === 'user'
   616→                          ? 'bg-primary text-primary-foreground'
   617→                          : 'bg-muted/60 border'
   618→                      }`}
   619→                    >
   620→                      <div className="text-sm whitespace-pre-wrap break-words leading-relaxed">
   621→                        {m.pending && !m.content ? (
   622→                          <Loader2 className="size-4 animate-spin" />
   623→                        ) : (
   624→                          m.role === 'assistant'
   625→                            ? renderCitationsInline(m.content)
   626→                            : m.content
   627→                        )}
   628→                      </div>
   629→                      {m.citations.length > 0 && (
   630→                        <div className="mt-3 pt-3 border-t border-border/40 space-y-2">
   631→                          <p className="text-[10px] uppercase tracking-wide text-muted-foreground font-semibold">
   632→                            Sources — click to jump
   633→                          </p>
   634→                          {m.citations.map((c, i) => (
   635→                            <div
   636→                              id={`cite-c${i + 1}`}
   637→                              key={c.chunkId}
   638→                              className="rounded-md bg-background border p-2 text-xs flex gap-2"
   639→                            >
   640→                              <div className="min-w-0 flex-1">
   641→                                <p className="font-medium line-clamp-1">
   642→                                  {c.videoTitle} · {formatTimestamp(c.startSec)}
   643→                                </p>
   644→                                <p className="text-muted-foreground line-clamp-2 mt-0.5">{c.preview}</p>
   645→                              </div>
   646→                              <Button
   647→                                variant="secondary"
   648→                                size="sm"
   649→                                className="h-7 text-[11px] flex-shrink-0"
   650→                                onClick={() => onCitationClick(c)}
   651→                                aria-label={`Jump to ${formatTimestamp(c.startSec)} in ${c.videoTitle}`}

   653→                                <Play className="size-3 mr-1" />
   654→                                Jump to {formatTimestamp(c.startSec)}
   655→                              </Button>
   656→                            </div>
   657→                          ))}
   658→                        </div>
   659→                      )}
   660→                    </div>
   661→                  </div>
   662→                ))
   663→              )}
   664→              <div ref={messagesEndRef} />
   665→            </div>
   666→          </ScrollArea>
   667→
   668→          {/* Input */}
   669→          <div className="border-t p-3 bg-background">
   670→            <div className="max-w-3xl mx-auto">
   671→              <div className="relative">
   672→                <Textarea
   673→                  ref={inputRef}
   674→                  value={input}
   675→                  onChange={(e) => setInput(e.target.value)}
   676→                  onKeyDown={onKeyDown}
   677→                  placeholder={videos.length ? 'Ask a question about the transcripts…' : 'Import a video first, then ask…'}

   682→                <Button
   683→                  size="icon"

   689→                  {streaming ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   690→                </Button>
   691→              </div>
   692→              <p className="text-[10px] text-muted-foreground mt-1.5 flex items-center gap-1">
   693→                <CornerDownLeft className="size-2.5" />
   694→                Enter to send · Shift+Enter for newline
   695→              </p>
   696→            </div>
   697→          </div>
   698→        </section>
   699→
   700→        {/* Right column: video player + chapters */}
   701→        <aside className="flex flex-col min-h-[40vh] lg:h-[calc(100vh-57px)] bg-muted/20">
   702→          {!activeVideo ? (
   703→            <div className="flex-1 grid place-items-center p-6 text-center">
   704→              <div>
   705→                <div className="size-12 mx-auto rounded-lg bg-muted grid place-items-center mb-3">
   706→                  <Play className="size-6 text-muted-foreground" />
   707→                </div>
   708→                <p className="text-sm font-medium">No video selected</p>
   709→                <p className="text-xs text-muted-foreground mt-1 max-w-[240px]">
   710→                  Click a video in the sidebar, or ask a question — citations will auto-focus the
   711→                  referenced video here.
   712→                </p>
   713→              </div>
   714→            </div>
   715→          ) : (
   716→            <>
   717→              <div className="aspect-video w-full bg-black">
   718→                <iframe
   719→                  ref={playerRef}
   720→                  key={`${activeVideo.youtubeId}-${activeVideo.startSec ?? 0}`}
   721→                  src={buildEmbedUrl(activeVideo.youtubeId, { startSec: activeVideo.startSec ?? 0, autoplay: false })}
   722→                  title={activeVideo.title}
   723→                  allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
   724→                  allowFullScreen
   725→                  className="w-full h-full"
   726→                />
   727→              </div>
   728→              <div className="p-3 border-b">
   729→                <p className="text-xs font-medium line-clamp-2 leading-snug">{activeVideo.title}</p>
   730→                <a
   731→                  href={`https://www.youtube.com/watch?v=${activeVideo.youtubeId}`}
   732→                  target="_blank"
   733→                  rel="noopener noreferrer"
   734→                  className="text-[10px] text-muted-foreground hover:text-foreground inline-flex items-center gap-1 mt-1"
   735→                >
   736→                  <ExternalLink className="size-2.5" />
   737→                  Open on YouTube
   738→                </a>
   739→              </div>
   740→              <div className="p-3 border-b">
   741→                <h3 className="text-xs font-semibold flex items-center gap-1.5 mb-2">
   742→                  <BookOpen className="size-3.5" />
   743→                  Chapters
   744→                </h3>
   745→              </div>
   746→              <ScrollArea className="flex-1">
   747→                <div className="p-2 space-y-0.5">
   748→                  {chaptersLoading ? (
   749→                    <>
   750→                      <Skeleton className="h-8 w-full" />
   751→                      <Skeleton className="h-8 w-full" />
   752→                      <Skeleton className="h-8 w-full" />
   753→                    </>
   754→                  ) : chapters.length === 0 ? (
   755→                    <p className="p-3 text-[11px] text-muted-foreground text-center">
   756→                      No chapters detected (video may be too short for gap-based chaptering).

   758→                  ) : (
   759→                    chapters.map((c) => (
   760→                      <button
   761→                        key={c.id}
   762→                        onClick={() => onChapterClick(c)}

   764→                      >
   765→                        <span className="text-[10px] text-muted-foreground font-mono tabular-nums flex-shrink-0 mt-0.5">
   766→                          {formatTimestamp(c.startSec)}
   767→                        </span>
   768→                        <span className="text-xs line-clamp-2 group-hover:text-foreground">
   769→                          {c.title}
   770→                        </span>
   771→                      </button>
   772→                    ))
   773→                  )}
   774→                </div>
   775→              </ScrollArea>
   776→            </>

   778→        </aside>
   779→      </main>
   780→
   781→      {/* Footer */}
   782→      <footer className="border-t bg-background px-4 py-2 text-[11px] text-muted-foreground text-center mt-auto">
   783→        YouTube Transcript RAG — local-first hybrid retrieval + streaming chat. Built with Next.js 16,

   786→    </div>
