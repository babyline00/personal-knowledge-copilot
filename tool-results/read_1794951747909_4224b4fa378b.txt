   116→function renderCitationsInline(content: string): React.ReactNode[] {
   117→  // Replace [^c1] markers with clickable chips linked to the citation block.
   118→  const parts: React.ReactNode[] = []
   119→  const re = /\[\^(c\d+)\]/g
   120→  let last = 0
   121→  let m: RegExpExecArray | null
   122→  while ((m = re.exec(content)) !== null) {
   123→    if (m.index > last) {
   124→      parts.push(<React.Fragment key={`t-${last}`}>{content.slice(last, m.index)}</React.Fragment>)
   125→    }
   126→    const id = m[1]
   127→    parts.push(
   128→      <a
   129→        key={`c-${m.index}`}
   130→        href={`#cite-${id}`}
   131→        className="text-primary text-xs align-super mx-0.5 px-1 rounded bg-primary/10 hover:bg-primary/20 no-underline"
   132→      >
   133→        [{id}]
   134→      </a>,
   135→    )
   136→    last = m.index + m[0].length
   137→  }
   138→  if (last < content.length) {
   139→    parts.push(<React.Fragment key={`t-${last}`}>{content.slice(last)}</React.Fragment>)
   140→  }
   141→  return parts
   142→}

   331→  // ---------- send chat ----------
   332→  async function handleSend() {
   333→    const text = composer.trim()
   334→    if (!text || !currentUserId || streaming) return
   335→    setComposer('')
   336→    setStreaming(true)
   337→    const userMsgId = `u-${Date.now()}`
   338→    const assistantMsgId = `a-${Date.now()}`
   339→    setMessages((prev) => [
   340→      ...prev,
   341→      { id: userMsgId, role: 'user', content: text, citations: [] },
   342→      { id: assistantMsgId, role: 'assistant', content: '', citations: [], pending: true },
   343→    ])
   344→
   345→    try {
   346→      const res = await fetch('/api/chat', {
   347→        method: 'POST',
   348→        headers: { 'Content-Type': 'application/json' },
   349→        body: JSON.stringify({ message: text, sessionId: sessionId ?? undefined, userId: currentUserId }),
   350→      })
   351→      if (!res.ok) {
   352→        const errText = await res.text()
   353→        throw new Error(errText || `HTTP ${res.status}`)
   354→      }
   355→      if (!res.body) throw new Error('No stream body')
   356→
   357→      const newSessionId = res.headers.get('X-Session-Id')
   358→      if (newSessionId) setSessionId(newSessionId)
   359→
   360→      const reader = res.body.getReader()
   361→      const decoder = new TextDecoder()
   362→      let buffer = ''
   363→      let fullMessage = ''
   364→      let denied = false
   365→      let sawCitationsMarker = false
   366→
   367→      while (true) {
   368→        const { done, value } = await reader.read()
   369→        if (done) break
   370→        buffer += decoder.decode(value, { stream: true })
   371→
   372→        // Special header: the server may prepend `---DENIED---\n` if safe-mode
   373→        // fired (RBAC denied the user).
   374→        if (!sawCitationsMarker && fullMessage === '' && buffer.startsWith('---DENIED---')) {
   375→          denied = true
   376→          const nl = buffer.indexOf('\n')
   377→          if (nl >= 0) {
   378→            buffer = buffer.slice(nl + 1)
   379→          } else {
   380→            buffer = ''
   381→          }
   382→        }
   383→
   384→        const splitIdx = buffer.indexOf('\n---CITATIONS---\n')
   385→        let textChunk = buffer
   386→        if (splitIdx >= 0) {
   387→          textChunk = buffer.slice(0, splitIdx)
   388→          sawCitationsMarker = true
   389→          // The citations JSON follows
   390→          const tail = buffer.slice(splitIdx + '\n---CITATIONS---\n'.length)
   391→          buffer = ''
   392→          fullMessage += textChunk
   393→          // Update message with accumulated text
   394→          setMessages((prev) =>
   395→            prev.map((m) =>
   396→              m.id === assistantMsgId ? { ...m, content: fullMessage, denied } : m,
   397→            ),
   398→          )
   399→          // Try parsing the trailing JSON to extract the citations array.
   400→          try {
   401→            const parsed = JSON.parse(tail) as { sessionId?: string; citations: Citation[]; denied?: boolean }
   402→            if (parsed.sessionId) setSessionId(parsed.sessionId)
   403→            if (typeof parsed.denied === 'boolean') denied = parsed.denied
   404→            setMessages((prev) =>
   405→              prev.map((m) =>
   406→                m.id === assistantMsgId ? { ...m, content: fullMessage, citations: parsed.citations, denied, pending: false } : m,
   407→              ),
   408→            )
   409→          } catch {
   410→            setMessages((prev) => prev.map((m) => (m.id === assistantMsgId ? { ...m, content: fullMessage, denied, pending: false } : m)))
   411→          }
   412→          break
   413→        } else {
   414→          fullMessage += textChunk
   415→          buffer = ''
   416→          setMessages((prev) =>
   417→            prev.map((m) => (m.id === assistantMsgId ? { ...m, content: fullMessage, denied } : m)),
   418→          )
   419→        }
   420→      }
   421→      // If the stream ended without a citations footer (LLM error / abort)
   422→      setMessages((prev) =>
   423→        prev.map((m) =>
   424→          m.id === assistantMsgId ? { ...m, content: m.content || '', denied, pending: false } : m,
   425→        ),
   426→      )
   427→      // Refresh admin stats — a new AuditLog row was just written.
   428→      refreshStats()
   429→    } catch (e) {
   430→      const err = `⚠️ ${(e as Error).message}`
   431→      setMessages((prev) =>
   432→        prev.map((m) =>
   433→          m.id === assistantMsgId ? { ...m, content: err, pending: false } : m,
   434→        ),
   435→      )
   436→      toast({ title: 'Chat error', description: (e as Error).message, variant: 'destructive' })
   437→    } finally {
   438→      setStreaming(false)
   439→    }
   440→  }
